### Task 4.1: Bilingual-parity test (TDD)

- [ ] **Step 1: Write the test** (it will pass once Phases 1-2 are committed):

```python
# tests/test_bilingual_docs.py
"""Parity: every English user-facing doc/notebook has a Spanish counterpart."""
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]  # package root


def test_readme_es_exists():
    assert (ROOT / "README.es.md").is_file()


def test_user_docs_have_spanish():
    en_docs = ["credentials", "cache_db", "connector_health", "signal_contract",
               "examples_gallery", "1.0_path", "lexicon_review"]
    missing = [d for d in en_docs if not (ROOT / "docs" / "es" / f"{d}.md").is_file()]
    assert not missing, f"missing docs/es/: {missing}"


def test_notebooks_have_spanish():
    nb = ROOT / "notebooks"
    en = [p for p in nb.glob("*.py") if not p.name.startswith("_")
          and not p.stem.endswith("_es")]
    missing = [p.name for p in en if not (nb / f"{p.stem}_es.py").is_file()]
    assert not missing, f"notebooks lacking _es sibling: {missing}"


def test_switcher_in_english_readme():
    txt = (ROOT / "README.md").read_text(encoding="utf-8")
    assert "Español" in txt[:600], "language switcher missing from README.md top"
```
